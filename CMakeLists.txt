cmake_minimum_required(VERSION 3.24)
project(FilmStation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Test)
qt_standard_project_setup()
add_library(filmstation_core src/core/modules.cpp src/modules/builtinmodules.cpp)
target_include_directories(filmstation_core PUBLIC src)
target_link_libraries(filmstation_core PUBLIC Qt6::Widgets)
add_library(filmstation_ui src/ui/mainwindow.cpp)
target_link_libraries(filmstation_ui PUBLIC filmstation_core)
qt_add_executable(FilmStation MACOSX_BUNDLE src/main.cpp)
target_link_libraries(FilmStation PRIVATE filmstation_ui)
set_target_properties(FilmStation PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER com.filmstation.desktop)
include(CTest)
if(BUILD_TESTING)
 qt_add_executable(filmstation_tests tests/workstation.cpp)
 target_link_libraries(filmstation_tests PRIVATE filmstation_ui Qt6::Test)
 add_test(NAME workstation COMMAND filmstation_tests)
 set_tests_properties(workstation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
